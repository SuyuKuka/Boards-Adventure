Format pour les JSON : 
Paramètres généraux absoluments nécessaire :
    "id" <- numéro de la case (0 pour le start, > 0 pour la suite des cases accessibles, < 0 pour les cellules annexes {Prison, Trône, ...}).
    "type" <- type de la case (PROPERTY, KINGDOM, CARD, TELEPORTATION, TAX, JAIL ou VOID).
    "xPos" <- position x du centre de la cellule sur le fichier png
    "yPos" <- position y du centre de la cellule sur le fichier png

Paramètres pour les PROPERTY :
    "group" <- id du groupe de la propriété (même couleur = même groupe).
	"values" <- liste de 4 cases contenant le prix d'achat et amélioration.

	{
        "id" : 4,
        "type" : "PROPERTY",
        "values" : [X, X, X, X],
        "group" : 1,
        "xPos" : 423,
        "yPos" : 423
    },

Paramètres pour les KINGDOM :
    "throneCell" <- id de la cellule du trône.

	{
        "id" : 5,
        "type" : "KINGDOM",
        "throneCell" : -2,
        "xPos" : 423,
        "yPos" : 302
    },

Paramètres pour les START :
    "lapGain" <- montant ajouté aux joueurs qui font un tour de la map.

	{
        "id" : 0,
        "type" : "START",
        "lapGain" : X,
        "xPos" : 60,
        "yPos" : 423
    },

Paramètres pour les CARD :
	"card" <- type de carte (ATTACK ou DEFENSE).

	{
        "id" : 12,
        "type" : "TELEPORTATION",
        "card" : DEFENSE,
        "xPos" : 423,
        "yPos" : 60
    },

Paramètres pour les TELEPORTATION :
	"targetCell" <- id de la case cible.

	{
        "id" : 12,
        "type" : "TELEPORTATION",
        "targetCell" : 2,
        "xPos" : 423,
        "yPos" : 60
    },



Paramètres pour les GAIN :
    "gain" <- montant (positif ou négatif) qu'il gagne quand il arrive sur cette case.

	{
        "id" : 5,
        "type" : "GAIN",
        "gain" : X,
        "xPos" : 302,
        "yPos" : 302
    },
